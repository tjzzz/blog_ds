    T3 三数之和#

    排序后，外面一层循环，里面是双指针

        def threeSum(self, nums):
        """
        :type nums: List[int]
        :rtype: List[List[int]]
        """
        if len(nums) < 3:
            return []

        ## 因为和是0，所以第一个数必然是负数
        n = len(nums)
        nums.sort()
        res = []
        for k in range(n):
            if nums[k] > 0:
                break
            i = k + 1
            j = n - 1 
            while i < j:
                ss = nums[k] + nums[i] + nums[j]
                if ss == 0:
                    ans = [nums[k], nums[i], nums[j]]
                    if ans not in res:
                        res.append(ans)
                    i += 1
                    j -= 1
                else:
                    if ss > 0:
                        j -= 1
                    else:
                        i += 1
        return res

        def FourSum(self, nums):
        n = len(nums)
        nums.sort()
        for i in range(n):
            for j in range(i+1, n):
                if nums[i] + nums[j] > 0:
                    return []

                k = j + 1
                m = n - 1 

                while k < m:
                    ss = nums[i] + nums[j] + nums[k] + nums[m]
                    if ss > 0:
                        m -= 1
                    else:
                        if ss < 0:
                            k += 1
                        else:
                            return [nums[i], nums[j], nums[k], nums[m]]
        return []

    T39-组合总和 T40-组合总和II#

    给你一个 无重复元素 的整数数组 candidates 和一个目标整数 target ，找出 candidates 中可以使数字和为目标数 target 的 所有 不同组合 ，并以列表形式返回。你可以按 任意顺序 返回这些组合。
    candidates 中的 同一个 数字可以 无限制重复被选取 。如果至少一个数字的被选数量不同，则两种组合是不同的。 

    官方解析：
    对于这类寻找所有可行解的题，我们都可以尝试用搜索回溯的方法来解决。

    回到本题，我们定义递归函数dfs(target,combine,idx) 表示当前在 candidates 数组的第 idx 位，还剩target 要组合，已经组合的列表为combine。
    递归的终止条件为 target≤0 或者 candidates 数组被全部用完。那么在当前的函数中，每次我们可以选择:
    - 跳过不用第 idx 个数，即执行 dfs(target,combine,idx+1)。
    - 也可以选择使用第idx 个数，即执行 dfs(target−candidates[idx], combine, idx)，注意到每个数字可以被无限制重复选取，因此搜索的下标仍为 idx。

    更形象化地说，如果我们将整个搜索过程用一个树来表达，即如下图呈现，每次的搜索都会延伸出两个分叉，直到递归的终止条件，这样我们就能不重复且不遗漏地找到所有可行解：

    # T39
def combinationSum(self, candidates, target):
    """
    :type candidates: List[int]
    :type target: int
    :rtype: List[List[int]]
    """
    res = []
    n = len(candidates)

    def dfs(i, tmp_list, tmp_sum):
        if i == n or tmp_sum > target:
            return 
        if tmp_sum == target:
            res.append(tmp_list)
            return 

        ## 如果选择第i位置的数字
        dfs(i, tmp_list + [candidates[i]], tmp_sum + candidates[i])
        dfs(i+1, tmp_list, tmp_sum)

    dfs(0, [], 0)

    return res

    def permuteUnique(self, nums):
    """
    :type nums: List[int]
    :rtype: List[List[int]]
    """
    nums.sort()  # 排序

    res = []
    visited = len(nums) * [0]
    n = len(nums)
    def dfs(tmp_list, visited):
        if sum(visited) == n:
            res.append(tmp_list)
            #return 
        for i in range(len(visited)):
            ## 判断重复
            if i > 0 and nums[i] == nums[i-1] and visited[i-1] == 0:
                continue
            if visited[i] == 0:
                visited[i] = 1
                dfs(tmp_list + [nums[i]], visited)
                visited[i] = 0  # 注意这里需要做回溯更新

    dfs([], visited)

    return res

    T33. 搜索旋转排序数组#

    class Solution(object):
    def search(self, nums, target):
        """
        :type nums: List[int]
        :type target: int
        :rtype: int
        """
        l, r = 0, len(nums) - 1

        while l <= r:
            mid = (l + r) // 2
            if nums[mid] == target:
                return mid

            if nums[mid] >= nums[l]:  # 左侧
                if nums[mid] < target or nums[l] > target:
                    l = mid + 1
                else:
                    r = mid -1
            else:
                if nums[mid] > target:
                    r = mid -1
                else:
                    if target >= nums[l]:
                        r = mid -1
                    else:
                        l = mid + 1


        return -1

    T11. 盛最多水的容器#

    给定一个长度为 n 的整数数组 height 。有 n 条垂线，第 i 条线的两个端点是 (i, 0) 和 (i, height[i]) 。找出其中的两条线，使得它们与 x 轴共同构成的容器可以容纳最多的水。

    class Solution(object):
    def maxArea(self, height):
        """
        :type height: List[int]
        :rtype: int
        """
        # s = (r-l) * min(height[r], height[l])
        l, r = 0, len(height)-1
        s_max = 0
        while l < r:
            s = (r - l) * min(height[l], height[r])
            if s > s_max:
                s_max = s
            # 找到高的,然后更新
            if height[l] < height[r]:
                l += 1
            else:
                r -= 1

        return s_max

    class Solution(object):
    def removeDuplicates(self, nums):
        """
        :type nums: List[int]
        :rtype: int
        """
        ## 数组已经排序
        j = 0
        for i in range(1, len(nums)):
            if nums[i] != nums[i - 1]:
                j += 1
            nums[j] = nums[i]
        return j + 1

    T53. 最大子数组和#

    动态规划
    给你一个整数数组 nums ，请你找出一个具有最大和的连续子数组（子数组最少包含一个元素），返回其最大和。子数组 是数组中的一个连续部分。

    思路：考虑以结尾的一个子数组，我们有f(i) = max(f(i-1) + nums[i], nums[i])

    def maxSubArray(self, nums):
    """
    :type nums: List[int]
    :rtype: int
    """

    ## 动态规划  f(i) = max(f(i-1) + nums[i], nums[i])
    n = len(nums)
    res = nums[0]
    for i in range(1, n):
        nums[i] = max(nums[i-1] + nums[i], nums[i])

        if nums[i] > res:
            res = nums[i]

    return res

    def firstMissingPositive(self, nums):
    """
    :type nums: List[int]
    :rtype: int
    """


    n = len(nums)
    default = n + 1

    if 1 not in nums:
        return 1

    for i in range(n):
        if nums[i] <= 0:
            nums[i] = default
    # 
    for i in range(n):
        abs_val = abs(nums[i])
        if abs_val < default:
            nums[abs_val-1] = -abs(nums[abs_val-1])   # 将原来的数变为负数
            #print(i, nums)
    #
    for i in range(n):
        if nums[i] > 0:
            return i+1
    return n+1

    class Solution:
    def solveNQueens(self, n: int) -> List[List[str]]:
        res = []
        def myfun(x, y, i, j):
            # 斜线
            if x -y == i - j:
                return 1
            if abs(x-i) == abs(y -j):
                return 1
            ## 
            if x== i or y == j:
                return 1
            return 0

        def dfs(tmp_list):
            ## 为了确保不重读，按照行的顺序开始筛选[0,x][1,x][2,x]
            if len(tmp_list) == n:
                res.append(tmp_list)
                return 

            tmp_len = len(tmp_list)
            i = tmp_len

            if i == 0:
                for j in range(n):
                    dfs([[i, j]])
            else:
                for j in range(n):
                    tmp_flag = 0
                    for item in tmp_list:
                        x, y = item
                        tmp_flag += myfun(x, y, i, j)
                    if tmp_flag == 0:
                        dfs(tmp_list + [[i, j]])
        #print(res)
        dfs([])
