def searchInsert(nums, target):

	"""
	:type nums: List[int]
	:type target: int
	:rtype: int
	"""
	if len(nums) == 0:
		return -1
		

	l = 0
	r = len(nums) - 1
	while l <= r:   # <=
		mid = (l + r) // 2    # 注意整除
		if nums[mid] == target:
			return mid
		
		if nums[mid] > target:
			r = mid - 1
		else:
			l = mid + 1
		## 最终插入位置是当前的下一个，此时l>r
	return -1
	# return l # 返回最接近的那个

def binarySearch(nums, target):

    :type target: int
    :rtype: int
    """

    if len(nums) == 0:
        return -1

    left, right = 0, len(nums)
    while left < right:
        mid = (left + right) // 2
        if nums[mid] == target:
            return mid
        elif nums[mid] < target:
            left = mid + 1
        else:
            right = mid
    # Post-processing:
    # End Condition: left == right
    if left != len(nums) and nums[left] == target:
        return left
    return -1

class Solution(object):
    def search(self, nums, target):

        """
        :type nums: List[int]
        :type target: int
        :rtype: int
        """

        l, r = 0, len(nums) - 1
        while l <= r:
            mid = (l + r) // 2
            if nums[mid] == target:
                return mid
            if nums[mid] >= nums[l]:  # 左侧

def binarySearch(nums, target):
    """
    :type nums: List[int]
    :type target: int
    :rtype: int
    """
    if len(nums) == 0:
        return -1

    left, right = 0, len(nums) - 1
    while left <= right:
        mid = (left + right) // 2
        if nums[mid] == target:
            return mid
        elif nums[mid] < target:
            left = mid + 1
        else:
            right = mid - 1

    # End Condition: left > right
    return -1
